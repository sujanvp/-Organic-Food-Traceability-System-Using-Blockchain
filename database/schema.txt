Table: products
CREATE TABLE products (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        product_name TEXT,
        product_type TEXT,
        farmer_id INTEGER,
        manufacturer_id INTEGER,
        status TEXT,
        certification_no TEXT,
        quantity REAL,
        unit TEXT,
        date_added TEXT,
        description TEXT,
        FOREIGN KEY (farmer_id) REFERENCES users(id),
        FOREIGN KEY (manufacturer_id) REFERENCES users(id)
    )
